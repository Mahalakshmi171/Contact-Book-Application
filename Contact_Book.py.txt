
import json
from pathlib import Path

FILE = Path("contacts.json")

if FILE.exists():
    contacts = json.loads(FILE.read_text())
else:
    contacts = {}

def save_contacts():
    FILE.write_text(json.dumps(contacts, indent=4))

while True:
    print("\nCONTACT BOOK")
    print("1. Add Contact")
    print("2. View Contacts")
    print("3. Search Contact")
    print("4. Exit")

    choice = input("Choose: ")

    if choice == "1":
        name = input("Name: ").strip()
        phone = input("Phone: ").strip()

        if name and phone:
            contacts[name] = phone
            save_contacts()
            print("Contact saved!")
        else:
            print("Name and phone are required.")

    elif choice == "2":
        for name, phone in contacts.items():
            print(name, ":", phone)
        if not contacts:
            print("No contacts saved.")

    elif choice == "3":
        name = input("Enter name to search: ").strip()
        print(name, ":", contacts.get(name, "Not found"))

    elif choice == "4":
        break

    else:
        print("Invalid choice.")
